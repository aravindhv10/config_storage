** Main script to run within the container
#+begin_src sh :shebang #!/bin/sh :results output :mkdirp yes :tangle ./container_init.sh
  compile_cvusability(){
      test -e '/root/.cache/model_cv_usability.pt2' \
          || 'python3' '/PYTHON_CODE/CV/compile.py' \
              '/data/input/cv_usability.ckpt' '/root/.cache/model_cv_usability.pt2' ;
  }

  compile_video(){
      test -e '/root/.cache/model_rd.pt2' \
          || 'python3' '/PYTHON_CODE/RD/compile_model.py' ;
  }

  build_cpp() {
      (test -e '/root/.cache/libmytorch.so' \
              && test -e '/root/.cache/libmytorch.so.1.0' \
              && cp '/root/.cache/libmytorch.so' '/lib/libmytorch.so' \
              && cp '/root/.cache/libmytorch.so.1.0' '/lib/libmytorch.so.1.0') \
          || ('/CPP_CODE/cmake_build.sh' \
              && cp -vf -- '/lib/libmytorch.so' '/root/.cache/libmytorch.so' \
              && cp -vf -- '/lib/libmytorch.so.1.0' '/root/.cache/libmytorch.so.1.0')
  }

  build_rust() {
      (test -e '/root/.cache/video_fft' \
              && test -e '/root/.cache/infer-client' \
              && test -e '/root/.cache/infer-server' \
              && cp -vf -- '/root/.cache/video_fft' '/bin/video_fft' \
              && cp -vf -- '/root/.cache/infer-client' '/bin/infer-client' \
              && cp -vf -- '/root/.cache/infer-server' '/bin/infer-server') \
          || ('/RUST_CODE/all_build.sh' && '/RUST_CODE/all_install.sh' \
              && cp -vf -- '/bin/video_fft' '/root/.cache/video_fft' \
              && cp -vf -- '/bin/infer-client' '/root/.cache/infer-client' \
              && cp -vf -- '/bin/infer-server' '/root/.cache/infer-server')
  }

  build_cpp
  build_rust
  compile_video
  compile_cvusability
  exec '/RUST_CODE/run_server.sh'
#+end_src

  pub fn get_hash_key_size() -> usize {
      std::mem::size_of::<key_type>()
  }

  pub struct blob_hash {
      pub hash: key_type,
      string_rep: std::string::String,
      filepath: Option<std::path::PathBuf>,
  }

  impl blob_hash {
      #[inline(always)]
      pub fn new_from_slice(inblob: &[u8]) -> Self {
          let hash = blake3::hash(inblob);
          let string_rep = c32::encode(hash.as_slice());
          Self {
              hash: hash,
              string_rep: string_rep,
              filepath: None,
          }
      }

      pub fn new_from_tmp_file(infilepath: impl AsRef<std::path::Path>) -> anyhow::Result<Self> {
          let file: std::fs::File = std::fs::File::open(infilepath.as_ref())?;
          let mmap_result = unsafe { memmapix::Mmap::map(&file) }?;
          let tmp = Self::new_from_slice(&mmap_result);
          Ok(tmp)
      }

      #[inline(always)]
      pub fn new_from_file(infilepath: impl AsRef<std::path::Path>) -> anyhow::Result<Self> {
          let mut tmp = Self::new_from_tmp_file(infilepath.as_ref())?;
          tmp.filepath = Some(infilepath.as_ref().to_path_buf());
          Ok(tmp)
      }

      #[inline(always)]
      pub fn get_hash(&self) -> &[u8] {
          self.hash.as_slice()
      }

      #[inline(always)]
      pub fn get_hash_string(&self) -> &str {
          &self.string_rep
      }

      #[inline(always)]
      pub fn get_file_path(&self) -> &Option<std::path::PathBuf> {
          &self.filepath
      }
  }

  pub struct VideoReader {
      pub opt: memfd::MemfdOptions,
  }

  pub struct VideoReaderPayload {
      pub data: memmapix::Mmap,
  }

  impl VideoReader {
      pub fn new() -> Self {
          Self {
              opt: memfd::MemfdOptions::default().allow_sealing(true),
          }
      }

      pub async fn tensor_from_slice_async(
          &self,
          data: &[u8],
          name: &str,
      ) -> anyhow::Result<VideoReaderPayload> {
          tracing::info!("Started processing video content in async mode.");
          let memfd = self.opt.create(name)?;
          let out_memfd = self.opt.create(name.to_string() + "out")?;
          tracing::info!("Created input and output memfds.");

          let mut res: usize = 0;
          while res < data.len() {
              res += rustix::io::write(memfd.as_file(), &data[res..])?;
          }

          memfd.as_file().seek(std::io::SeekFrom::Start(0))?;
          tracing::info!("Wrote data into the FD and seeked to start.");

          match ffmpeg_video_to_raw_stdin_stdout_async(
              std::process::Stdio::from(memfd.as_file().try_clone()?),
              std::process::Stdio::from(out_memfd.as_file().try_clone()?),
          )
          .await?
          .code()
          {
              Some(i) => {
                  if i != 0 {
                      tracing::error!("FFMPEG exited with non-0 error code {}", i);
                      return Err(anyhow::format_err!("ffmpeg failed with error code {}", i));
                  } else {
                      let memmap = tokio::task::spawn_blocking(move || unsafe {
                          memmapix::Mmap::map(out_memfd.as_file())
                      })
                      .await??;
                      tracing::info!("FFMPEG completed and final memfd is memory mapped.");
                      return Ok(VideoReaderPayload { data: memmap });
                  }
              }
              None => {
                  tracing::error!("FFMPEG exited with unknown error.");
                  return Err(anyhow::format_err!("ffmpeg failed with unknown error code"));
              }
          };
      }

  pub const IDEAL_LENGTH: u16 = (export::IDEAL_LENGTH_SECONDS_VIDEO * export::FPS_VIDEO_TENSOR) as u16;

  #[inline(always)]
  pub fn get_num_windows(total_video_length: u64) -> u8 {
      const MIN_LENGTH: u16 = (IDEAL_LENGTH * 3) >> 2;

      if total_video_length < (MIN_LENGTH as u64) {
          return 0 as u8;
      } else if total_video_length <= (IDEAL_LENGTH as u64) {
          return 1 as u8;
      } else {
          const STRIDE: u16 = IDEAL_LENGTH;

          let float_val =
              (((total_video_length - (IDEAL_LENGTH as u64)) as f64) / (STRIDE as f64)) as f64;

          let floor_val = float_val.floor();

          let diff = float_val - floor_val;

          let num_windows: u8 = {
              if diff < 0.1 {
                  (floor_val as u8) + 1
              } else {
                  (floor_val as u8) + 2
              }
          };

          return num_windows;
      }
  }

          let result = tokio::task::spawn_blocking(move || {

          })

      pub fn get_video_tensor(&self) -> anyhow::Result<&[export::frame_video_Y]> {
          if self.size_t < 2 {
              return Err(anyhow::format_err!("The video blob seems too small"));
          }

          let tensor_data = unsafe {
              std::slice::from_raw_parts(
                  self.mmap.as_ptr() as *const export::frame_video_Y,
                  self.size_t as usize,
              )
          };

          return Ok(tensor_data);
      }

      pub fn get_windowed_slices(&self) -> anyhow::Result<Vec<export::full_video>> {
          let num_windows = get_num_windows(/*total_video_length: u64 =*/ self.size_t as u64);

          tracing::info!("Need to split input to {} windows", num_windows);

          match num_windows {
              0 => {
                  tracing::error!("Video too short...");
                  return Err(anyhow::format_err!("Video too short..."));
              }
              1 => {
                  let mut ret: Vec<export::full_video> = Vec::with_capacity(1);
                  ret.push(export::full_video {
                      data: self.mmap.as_ptr() as *const export::frame_video_Y,
                      len_t: self.size_t as u32,
                  });

                  return Ok(ret);
              }
              _ => {
                  let calc = self.size_t / (num_windows as u16);
                  let len_chunk = {
                      if (self.size_t % (num_windows as u16)) == 0 {
                          calc as usize
                      } else {
                          (calc + 1) as usize
                      }
                  }
                  .max(IDEAL_LENGTH as usize);

                  tracing::info!("Length of each chunk = {}", len_chunk);

                  let dataptr = self.mmap.as_ptr() as *const export::frame_video_Y;

                  let mut ret: Vec<export::full_video> = Vec::with_capacity(num_windows as usize);

                  for i in 0..num_windows {
                      let video_end_pos = (len_chunk as usize)
                          + (((i as usize) * ((self.size_t as usize) - (len_chunk as usize)))
                              / ((num_windows as usize) - (1 as usize)));

                      let video_start_pos = video_end_pos - len_chunk;

                      tracing::debug!("Current chunk: {} - {}", video_start_pos, video_end_pos);

                      ret.push(export::full_video {
                          data: unsafe { dataptr.add(video_start_pos) },
                          len_t: len_chunk as u32,
                      });
                  }

                  return Ok(ret);
              }
          }
      }

  async fn infer_video_end_2_end(
      key: hasher::blob_hash,
      use_gpu: bool,
  ) -> anyhow::Result<Vec<export::rd_classifier_output>> {
      let filestore = filestore::file_store::new().await?;
      let raw_tensor = filestore.get_raw_tensor_async(&key).await?;

          let mut image_from_video: Vec<export::CV_USABILITY_IMAGE> = Vec::with_capacity(16);
          let image_from_video_ptr = image_from_video.as_mut_ptr();
          for i in 0..16 {
              let y = i * y2 / 15;
              let src = &video_tensor[y] as *const export::frame_video_Y;
              unsafe { export::frame_video_to_cv_usability(src, image_from_video_ptr.add(i)) };
          }
          unsafe { image_from_video.set_len(16) };
          let mut slave = inferencerelatedimagecvusability::infer_slave::new();
          let res = slave.infer(&mut image_from_video, inferencerelatedimagecvusability::GOOD_BATCH_SIZE)?;
          tracing::trace!("{:?}", res);
      }

      let mut list_video_fft_tensor =
          export::fft_video::windowed_from_video_slicer(&slicer, use_gpu)?;

      /* Normalize the video tensor */
      {
          let normalizer = videofftstats::fft_video_normalizer::new(
              /*path_file_bin64_mu: String =*/ "/data/input/train_mean.64bin",
              /*path_file_bin64_sigma: String =*/ "/data/input/train_sigma.64bin",
          )?;

          normalizer.normalize_vec(
              /*x: &mut Vec<videofft::fft_video> =*/ &mut list_video_fft_tensor,
          );
      }

      let mut infer_slave = inferencerelated::infer_slave::new();

      let ret = infer_slave.infer(
          &mut list_video_fft_tensor,
          inferencerelated::GOOD_BATCH_SIZE,
      )?;

      for i in ret.iter() {
          tracing::trace!("{} {} {}", i.out[0], i.out[1], i.out[2]);

          let mut senders: Vec<message_output_sender> = Vec::new();
          let mut do_loop = true;
          let mut do_infer = false;

          if true {
              let input_msg = receiver.recv();

              match input_msg {
                  Ok(o) => {
                      tensors.push(o.tensor);
                      senders.push(o.oneshot_send_channel);
                      do_loop = tensors.len() <= 8;
                      do_infer = true;
                      tracing::info!("Received and construct the 1st inference request");
                  }

                  Err(e) => {
                      tracing::error!(
                          "Failed to send inference request to the serving thread due to {}",
                          e
                      );
                      do_loop = false;
                      do_infer = false;
                  }
              }
          }

          let start = tokio::time::Instant::now();
          while do_loop {
              tracing::info!("Checking for additional requests");
              let message_input = receiver.recv_timeout(std::time::Duration::from_millis(100));

              match message_input {
                  Ok(o) => {
                      tensors.push(o.tensor);
                      senders.push(o.oneshot_send_channel);
                      do_loop = tensors.len() <= 8;
                      do_infer = true;
                      tracing::info!("Appended another batch to the request");
                  }
                  Err(_e) => {
                      do_loop = start.elapsed() < max_delay;
                  }
              }
          }

          if do_infer {
              let res = efficient_infer(tensors, &mut infer_slave);

              for (x, y) in res.into_iter().zip(senders.into_iter()) {
                  match x {
                      Ok(o) => {
                          tracing::info!("Inference succeeded, sending results");
                          y.send(o);
                      }
                      Err(e) => {
                          tracing::error!("Image inference for batch failed due to {}", e);
                          drop(y);
                      }
                  }
              }
          }
      }
  }

  async fn inference_loop(receiver: message_input_receiver) -> anyhow::Result<()> {
      let res = tokio::task::spawn_blocking(move || inference_loop_blocking(receiver)).await?;
      return res;
  }

  pub struct client {
      sender: message_input_sender,
      handle: Option<tokio::task::JoinHandle<anyhow::Result<()>>>,
  }

  impl Drop for client {
      fn drop(&mut self) {
          match self.handle.take() {
              Some(o) => {
                  o.abort();
              }
              None => {}
          }
      }
  }

  impl client {
      pub async fn new() -> std::sync::Arc<Self> {
          let (sender, receiver) = get_message_input_pair();
          let tmp = inference_loop(receiver);
          let handle = tokio::task::spawn(tmp);

          return std::sync::Arc::new(Self {
              sender: sender,
              handle: Some(handle),
          });
      }

      pub async fn do_infer_on_image_batch_tensor_async(
          &self,
          tensors_input: batched_input,
      ) -> anyhow::Result<Vec<f32>> {
          let (sender, receiver) = get_message_output_pair();

          let msg = message_input {
              tensor: tensors_input,
              oneshot_send_channel: sender,
          };

          match self.sender.send(msg).await {
              Ok(_) => {
                  tracing::info!("Successfullt sent request for inference");
              }
              Err(_) => {
                  tracing::error!("Failed to send request for inference");
                  return Err(anyhow::format_err!("Failed to send request for inference"));
              }
          };

          let ret = receiver.recv_async().await?;

          return Ok(ret);
      }
  }

      const image_FPS: u32 = 2;
      let ret = {
          let exact_num_batches = ((image_FPS * num_frames) as f32)

      pub async fn do_infer_on_video_file(
          &self,
          key: &hasher::blob_hash,
      ) -> anyhow::Result<combined_results> {
          tracing::debug!("Reading the video file using ffmpeg");

          let slicer = {
              let mmap = self.file_store.get_raw_tensor_async(&key).await?;
              videoview::video_slicer_mapped::new(mmap)
          }?;

          let image_infer_results = {
              let image_batch_tensor: Vec<export::CV_USABILITY_IMAGE> = {
                  let videotensor = slicer.get_video_tensor()?;

                  let total_images =
                      get_num_batches_in_video(/*num_frames: u32 =*/ slicer.size_t as u32)
                          ,* (serverinferencechannelimagecvusability::GOOD_BATCH_SIZE as u32);

                  let mut ret: Vec<export::CV_USABILITY_IMAGE> =
                      Vec::with_capacity(total_images as usize);

                  let ret_ptr = ret.as_mut_ptr();

                  for i in 0..total_images {
                      // (v2 - v1) / (i2 - i1) = (v - v1) / (i - i1)
                      let v = ((slicer.size_t as usize) * (i as usize)) / (total_images as usize);

                      unsafe {
                          export::frame_video_to_cv_usability(
                              /*src: *const frame_video_Y =*/
                              &videotensor[v] as *const export::frame_video_Y,
                              /*dst: *mut CV_USABILITY_IMAGE =*/
                              ret_ptr.add(i as usize),
                          )
                      };
                  }

                  unsafe { ret.set_len(total_images as usize) };

                  ret
              };

              tracing::debug!("Constructed the image tensor.");

              self.infpairimage
                  .do_infer_on_image_batch_tensor_async(image_batch_tensor)
              // .await?
          };

          tracing::debug!("Ran image inference.");

          let rd_infer_results = async move {
              let list_video_fft_tensor = tokio::task::spawn_blocking(move || {
                  export::fft_video::windowed_from_video_slicer(
                      &slicer, /*use_gpu: bool =*/ true,
                  )
              })
              .await??;

              self.infer_rd.do_infer_on_fft_tensor(list_video_fft_tensor)
              .await
          };

          let (image_infer_results, rd_infer_results) =
              tokio::join!(image_infer_results, rd_infer_results);

          tracing::debug!("Done video inference.");

          return Ok(combined_results {
              results_video: rd_infer_results?,
              results_image: image_infer_results?,
          });
      }

#+begin_src cpp :mkdirp yes :tangle ./src/file_mlock.hpp
  #ifndef _HEADER_GUARD_src_file_mlock_hpp_
  #define _HEADER_GUARD_src_file_mlock_hpp_

  #include "./includes.dir/header_os_c.hpp"
  #include "./main.hpp"

  #define _MACRO_SELF_ file_mlock

  class _MACRO_SELF_ {
  private:
    void *addr;
    i64_t length;

  public:
    _MACRO_SELF_(char const *path_file_input) : addr(MAP_FAILED), length(-1) {
      int fd = open(path_file_input, O_RDONLY);
      if (fd == -1) {
        printf("Failed to open the file %s\n", path_file_input);
      } else {
        struct stat st;
        if (fstat(fd, &st) == 0) {
          length = st.st_size;
          printf("Size of file %s is %ld\n", path_file_input, length);
          if (length > 0) {
            addr = mmap(NULL, length, PROT_READ, MAP_SHARED, fd, 0);
            if (addr == MAP_FAILED) {
              printf("mmap of file %s failed\n", path_file_input);
              length = -1;
            } else {
              if (mlock(addr, length) == 0) {
                printf("mlock for %s and length %ld succeeded\n", path_file_input,
                       length);
              } else {
                printf("mlock for %s failed\n", path_file_input);
                munmap(addr, length);
                addr = MAP_FAILED;
                length = -1;
              }
            }
          } else {
            printf("Got invalid file size %ld for file %s\n", length,
                   path_file_input);
            length = -1;
          }
        } else {
          printf("fstat for %s failed\n", path_file_input);
        }
        close(fd);
      }
    }

    ~_MACRO_SELF_() {
      if ((addr != MAP_FAILED) && (length > 0)) {
        munmap(addr, length);
      }
    }
  };

  #undef _MACRO_SELF_

  #endif

  #define _MACRO_SELF_ infer_slave

  class _MACRO_SELF_ {
  private:
    c10::InferenceMode mode;
    torch::TensorOptions options_input;
    torch::TensorOptions options_compute;
    torch::TensorOptions options_output;
    torch::inductor::AOTIModelPackageLoader loader;

  public:
    inline i32_t infer(CV_USABILITY_IMAGE const *blob_source,
                       CV_USABILITY_INFERENCE_OUTPUT *blob_destination,
                       u32_t const batch_size) {

      try {

        torch::Tensor cpu_tensor =
            torch::from_blob(const_cast<CV_USABILITY_IMAGE *>(blob_source),
                             {static_cast<i64_t>(batch_size), SIZE_IMAGE_Y,
                              SIZE_IMAGE_X, SIZE_IMAGE_C},
                             options_input);

        lock_guard tmp;

        std::vector<torch::Tensor> inputs = {cpu_tensor.to(options_compute)};

        std::vector<torch::Tensor> outputs = loader.run(inputs);

        torch::Tensor out_tensor = outputs[0].to(options_output).contiguous();

        std::memcpy(blob_destination, out_tensor.const_data_ptr(),
                    batch_size * sizeof(CV_USABILITY_INFERENCE_OUTPUT));

        return 0;

      } catch (const std::exception &e) {

        std::memset(blob_destination, 0,
                    batch_size * sizeof(CV_USABILITY_INFERENCE_OUTPUT));

        printf("Error: %s\n", e.what());

        return -1;
      }
    }

    _MACRO_SELF_(std::string const path_file_model)
        : options_input(get_input_device_and_dtype()),
          options_compute(get_inference_device_and_dtype()),
          options_output(get_output_device_and_dtype()), loader(path_file_model) {
    }

    ~_MACRO_SELF_() {}

    inline static _MACRO_SELF_ *NEW() {
      std::string path_file_model(get_compiled_model_path());
      return new _MACRO_SELF_(path_file_model);

    torch::Tensor intensor =
        torch::nn::functional::interpolate(
            torch::from_blob(
                const_cast<void *>(static_cast<void const *>(src)),
                {SIZE_VIDEO_TENSOR_Y, SIZE_VIDEO_TENSOR_X, SIZE_VIDEO_TENSOR_C},
                {DIST_VIDEO_TENSOR_Y, DIST_VIDEO_TENSOR_X, DIST_VIDEO_TENSOR_C},
                torch::TensorOptions()
                    .dtype(get_tensor_dtype<TYPE_DATA_VIDEO_TENSOR>())
                    .device(torch::kCPU))
                .narrow(1, (SIZE_VIDEO_TENSOR_X - SIZE_VIDEO_TENSOR_Y) >> 1,
                        SIZE_VIDEO_TENSOR_Y)
                .permute({2, 0, 1})
                .unsqueeze(0),
            torch::nn::functional::InterpolateFuncOptions()
                .size(std::vector<i64_t>({SIZE_IMAGE_Y, SIZE_IMAGE_X}))
                .mode(torch::kBilinear)
                .align_corners(false)) /* Done interpolating */
            .squeeze()
            .clamp(0, 255)
            .to(get_tensor_dtype<TYPE_DATA_IMAGE>())
            .permute({1, 2, 0})
            .contiguous();
    // Y1 X2 C0

    std::memcpy(dst, intensor.const_data_ptr(), sizeof(CV_USABILITY_IMAGE));
  }

  i32_t do_fft_compress(full_video const src, fft_video *const dest,
                        bool use_gpu) {

    u32_t constexpr position_start_x =
        (SIZE_VIDEO_TENSOR_X - SIZE_FFT_TENSOR_X) >> 1;

    u32_t constexpr position_start_y =
        (SIZE_VIDEO_TENSOR_Y - SIZE_FFT_TENSOR_Y) >> 1;

    u32_t constexpr position_end_x = position_start_x + SIZE_FFT_TENSOR_X;
    u32_t constexpr position_end_y = position_start_y + SIZE_FFT_TENSOR_Y;

    try {

      c10::InferenceMode mode;

      auto device_gpu =
          torch::cuda::is_available() && use_gpu ? torch::kCUDA : torch::kCPU;

      auto t_cutoff =
          torch::sum((torch::fft::rfftfreq(src.len_t,
                                           torch::TensorOptions()
                                               .dtype(get_tensor_dtype<f32_t>())
                                               .device(torch::kCPU)) *
                      FPS_VIDEO_TENSOR) /* Scaled frequency mode range tensor */
                     < FREQUENCY_LIMIT_FFT_TENSOR) /* Done evaluating number of
                                                      modes to keep */
              .item()
              .to<u32_t>();

      lock_guard tmp;

      torch::Tensor tensor_video_padded =
          torch::from_blob(
              /* data = */ const_cast<void *>(
                  static_cast<void const *>(src.data)),
              /* sizes = */
              {src.len_t, SIZE_VIDEO_TENSOR_Y, SIZE_VIDEO_TENSOR_X,
               SIZE_VIDEO_TENSOR_C},
              /* strides = */
              {DIST_VIDEO_TENSOR_T, DIST_VIDEO_TENSOR_Y, DIST_VIDEO_TENSOR_X,
               DIST_VIDEO_TENSOR_C},
              /* Device_DType = */
              torch::TensorOptions()
                  .dtype(get_tensor_dtype<u8_t>())
                  .device(torch::kCPU)) /* T Y X C */
              .to(device_gpu)
              .to(torch::kFloat32)
              .permute({3, 1, 2, 0}); // C Y X T

      if (true) {
        i64_t constexpr i = 3;

        tensor_video_padded =
            torch::fft::rfft(tensor_video_padded, /*n=*/std::nullopt, /*dim=*/i)
                .narrow(i, 0, t_cutoff);
      }

      if (true) {
        i64_t constexpr i = 2;

        tensor_video_padded =
            torch::fft::fftshift(
                torch::fft::fft(tensor_video_padded, std::nullopt, /*dim=*/i),
                {i})
                .narrow(i, position_start_x, SIZE_FFT_TENSOR_X);
      }

      if (true) {
        i64_t constexpr i = 1;

        tensor_video_padded =
            torch::fft::fftshift(
                torch::fft::fft(tensor_video_padded, std::nullopt, /*dim=*/i),
                {i})
                .narrow(i, position_start_y, SIZE_FFT_TENSOR_Y);
      }

      if (true) {
        i64_t constexpr i = 0;

        tensor_video_padded = torch::fft::fftshift(
            torch::fft::fft(tensor_video_padded, std::nullopt, /*dim=*/i), {i});
      }

      torch::Tensor compressed_tensor_video_fft =
          torch::nn::functional::interpolate(
              torch::cat(
                  {tensor_video_padded.abs(), tensor_video_padded.angle()},
                  /*dim=*/0) /* Done extracting abs and angle into real tensor*/
                  .unsqueeze(0),
              torch::nn::functional::InterpolateFuncOptions()
                  .size(
                      std::vector<i64_t>({SIZE_FFT_TENSOR_Y, SIZE_FFT_TENSOR_X,
                                          static_cast<i64_t>(SIZE_FFT_TENSOR_T)}))
                  .mode(torch::kTrilinear)
                  .align_corners(false)) /* Done interpolating */
              .squeeze()
              .to(torch::kCPU)
              .contiguous();

      std::memcpy(dest, compressed_tensor_video_fft.data_ptr(),
                  sizeof(fft_video));

      return 0;

    } catch (const std::exception &e) {
      std::memset(dest, 0, sizeof(fft_video));
      printf("Error: %s\n", e.what());
      return -1;
    }
  }

  class _MACRO_SELF_ {
  private:
    c10::InferenceMode mode;

    torch::TensorOptions options_input;
    torch::TensorOptions options_compute;
    torch::TensorOptions options_output;

    torch::inductor::AOTIModelPackageLoader loader;

  public:
    inline i32_t infer(fft_video const *blob_source,
                       rd_classifier_output *blob_destination,
                       u32_t const batch_size) {
      try {
        lock_guard tmp;

        torch::Tensor cpu_tensor = torch::from_blob(
            const_cast<void *>(static_cast<void const *>(blob_source)),
            {static_cast<i64_t>(batch_size),
             static_cast<i64_t>(SIZE_FFT_TENSOR_C),
             static_cast<i64_t>(SIZE_FFT_TENSOR_Y),
             static_cast<i64_t>(SIZE_FFT_TENSOR_X),
             static_cast<i64_t>(SIZE_FFT_TENSOR_T)},
            options_input);

        std::vector<torch::Tensor> inputs = {cpu_tensor.to(options_compute)};

        std::vector<torch::Tensor> outputs = loader.run(inputs);

        torch::Tensor out_tensor = outputs[0].to(options_output).contiguous();

        std::memcpy(blob_destination, out_tensor.const_data_ptr(),
                    batch_size * sizeof(rd_classifier_output));

        return 0;

      } catch (const std::exception &e) {

        std::memset(blob_destination, 0,
                    batch_size * sizeof(rd_classifier_output));

        printf("Error: %s\n", e.what());

        return 1;
      }
    }

    _MACRO_SELF_(std::string const path_file_model)
        : options_input(get_input_device_and_dtype()),
          options_compute(get_inference_device_and_dtype()),
          options_output(get_output_device_and_dtype()), loader(path_file_model) {
    }

    ~_MACRO_SELF_() {}

    inline static _MACRO_SELF_ *NEW() {
      std::string path_file_model(get_compiled_model_path());
      return new _MACRO_SELF_(path_file_model);
    }
  };

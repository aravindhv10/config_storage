  impl blob_hash {
      #[inline(always)]
      pub fn new_from_slice(inblob: &[u8]) -> Self {
          let hash = blake3::hash(inblob);
          let string_rep = c32::encode(hash.as_slice());
          Self {
              hash: hash,
              string_rep: string_rep,
              filepath: None,
          }
      }

      pub fn new_from_tmp_file(infilepath: impl AsRef<std::path::Path>) -> anyhow::Result<Self> {
          let file: std::fs::File = std::fs::File::open(infilepath.as_ref())?;
          let mmap_result = unsafe { memmapix::Mmap::map(&file) }?;
          let tmp = Self::new_from_slice(&mmap_result);
          Ok(tmp)
      }

      #[inline(always)]
      pub fn new_from_file(infilepath: impl AsRef<std::path::Path>) -> anyhow::Result<Self> {
          let mut tmp = Self::new_from_tmp_file(infilepath.as_ref())?;
          tmp.filepath = Some(infilepath.as_ref().to_path_buf());
          Ok(tmp)
      }

      #[inline(always)]
      pub fn get_hash(&self) -> &[u8] {
          self.hash.as_slice()
      }

      #[inline(always)]
      pub fn get_hash_string(&self) -> &str {
          &self.string_rep
      }

      #[inline(always)]
      pub fn get_file_path(&self) -> &Option<std::path::PathBuf> {
          &self.filepath

  impl VideoReader {
      pub fn new() -> Self {

      pub async fn tensor_from_slice_async(
          &self,
          data: &[u8],
          name: &str,
      ) -> anyhow::Result<VideoReaderPayload> {
          tracing::info!("Started processing video content in async mode.");
          let memfd = self.opt.create(name)?;
          let out_memfd = self.opt.create(name.to_string() + "out")?;
          tracing::info!("Created input and output memfds.");

          let mut res: usize = 0;
          while res < data.len() {
              res += rustix::io::write(memfd.as_file(), &data[res..])?;
          }

          memfd.as_file().seek(std::io::SeekFrom::Start(0))?;
          tracing::info!("Wrote data into the FD and seeked to start.");

          match ffmpeg_video_to_raw_stdin_stdout_async(
              std::process::Stdio::from(memfd.as_file().try_clone()?),
              std::process::Stdio::from(out_memfd.as_file().try_clone()?),
          )
          .await?
          .code()
          {
              Some(i) => {
                  if i != 0 {
                      tracing::error!("FFMPEG exited with non-0 error code {}", i);
                      return Err(anyhow::format_err!("ffmpeg failed with error code {}", i));
                  } else {
                      let memmap = tokio::task::spawn_blocking(move || unsafe {
                          memmapix::Mmap::map(out_memfd.as_file())
                      })
                      .await??;
                      tracing::info!("FFMPEG completed and final memfd is memory mapped.");
                      return Ok(VideoReaderPayload { data: memmap });
                  }
              }
              None => {
                  tracing::error!("FFMPEG exited with unknown error.");
                  return Err(anyhow::format_err!("ffmpeg failed with unknown error code"));
              }
          };
      }

      pub fn put_content_blocking(
          &self,
          key: &hasher::blob_hash,
          value: &[u8],
      ) -> anyhow::Result<()> {
          let name = key.get_hash_string();
          let key = key.hash.clone();

          match self.main_store.entry_sync(key) {
              scc::hash_map::Entry::Occupied(_) => {
                  tracing::warn!("Key {} already exists, not doing anything.", name);
              }
              scc::hash_map::Entry::Vacant(v) => {
                  let memfd = self.video_reader.opt.create(name)?;
                  let mut res: usize = 0;
                  while res < value.len() {
                      res += rustix::io::write(memfd.as_file(), &value[res..])?;
                  }
                  memfd.as_file().seek(std::io::SeekFrom::Start(0))?;
                  memfd.add_seal(memfd::FileSeal::SealWrite)?;
                  v.insert_entry(memfd);
              }
          };

          Ok(())
      }

  use crate::export;
  use crate::hasher;
  use crate::videodecoder;

  pub const IDEAL_LENGTH: u16 = (export::IDEAL_LENGTH_SECONDS_VIDEO * export::FPS_VIDEO_TENSOR) as u16;

  #[inline(always)]
  pub fn get_num_windows(total_video_length: u64) -> u8 {
      const MIN_LENGTH: u16 = (IDEAL_LENGTH * 3) >> 2;

      if total_video_length < (MIN_LENGTH as u64) {
          return 0 as u8;
      } else if total_video_length <= (IDEAL_LENGTH as u64) {
          return 1 as u8;
      } else {
          const STRIDE: u16 = IDEAL_LENGTH;

          let float_val =
              (((total_video_length - (IDEAL_LENGTH as u64)) as f64) / (STRIDE as f64)) as f64;

          let floor_val = float_val.floor();

          let diff = float_val - floor_val;

          let num_windows: u8 = {
              if diff < 0.1 {
                  (floor_val as u8) + 1
              } else {
                  (floor_val as u8) + 2
              }
          };

          return num_windows;
      }
  }

  pub struct video_slicer_mapped {
      pub size_t: u16,
      mmap: memmapix::Mmap,
  }

  impl video_slicer_mapped {
      pub fn new(mmap: memmapix::Mmap) -> anyhow::Result<Self> {
          let size_t: u16 = (mmap.len() / (export::DIST_VIDEO_TENSOR_T as usize)) as u16;

          if mmap.len() % (export::DIST_VIDEO_TENSOR_T as usize) != 0 {
              tracing::warn!(
                  "Mmap size {} is not an exact multiple of frame size {}",
                  mmap.len(),
                  export::DIST_VIDEO_TENSOR_T
              );
          }

          Ok(Self {
              size_t: size_t,
              mmap: mmap,
          })
      }

      async fn new_from_key_value(
          key: &hasher::blob_hash,
          value: std::sync::Arc<Vec<u8>>,
      ) -> anyhow::Result<Self> {
          let slave = videodecoder::VideoReader::new();
          let name = key.get_hash_string().to_string();
          let result = tokio::task::spawn_blocking(move || {
              slave.tensor_from_slice_blocking(&value, name.as_str())
          })
          .await??;

          Self::new(result.data)
      }

      pub fn get_video_tensor(&self) -> anyhow::Result<&[export::frame_video_Y]> {
          if self.size_t < 2 {
              return Err(anyhow::format_err!("The video blob seems too small"));
          }

          let tensor_data = unsafe {
              std::slice::from_raw_parts(
                  self.mmap.as_ptr() as *const export::frame_video_Y,
                  self.size_t as usize,
              )
          };

          return Ok(tensor_data);
      }

      pub fn get_windowed_slices(&self) -> anyhow::Result<Vec<export::full_video>> {
          let num_windows = get_num_windows(/*total_video_length: u64 =*/ self.size_t as u64);

          tracing::info!("Need to split input to {} windows", num_windows);

          match num_windows {
              0 => {
                  tracing::error!("Video too short...");
                  return Err(anyhow::format_err!("Video too short..."));
              }
              1 => {
                  let mut ret: Vec<export::full_video> = Vec::with_capacity(1);
                  ret.push(export::full_video {
                      data: self.mmap.as_ptr() as *const export::frame_video_Y,
                      len_t: self.size_t as u32,
                  });

                  return Ok(ret);
              }
              _ => {
                  let calc = self.size_t / (num_windows as u16);
                  let len_chunk = {
                      if (self.size_t % (num_windows as u16)) == 0 {
                          calc as usize
                      } else {
                          (calc + 1) as usize
                      }
                  }
                  .max(IDEAL_LENGTH as usize);

                  tracing::info!("Length of each chunk = {}", len_chunk);

                  let dataptr = self.mmap.as_ptr() as *const export::frame_video_Y;

                  let mut ret: Vec<export::full_video> = Vec::with_capacity(num_windows as usize);

                  for i in 0..num_windows {
                      let video_end_pos = (len_chunk as usize)
                          + (((i as usize) * ((self.size_t as usize) - (len_chunk as usize)))
                              / ((num_windows as usize) - (1 as usize)));

                      let video_start_pos = video_end_pos - len_chunk;

                      tracing::debug!("Current chunk: {} - {}", video_start_pos, video_end_pos);

                      ret.push(export::full_video {
                          data: unsafe { dataptr.add(video_start_pos) },
                          len_t: len_chunk as u32,
                      });
                  }

                  return Ok(ret);
              }
          }
      }
  }

          tracing::debug!("Passing the box pointer to torch tensor");
          if true {

              let video_view = crate::videoview::video_slicer_mapped::new(video_tensor.data)?;
              let windowed_slices = video_view.get_windowed_slices()?;

              let windowed_slices2 = windowed_slices.clone();

              let (a, b) = tokio::join!(
                  self.do_rd_infer(windowed_slices),
                  self.do_image_cv_usability_infer(windowed_slices2)
              );

              return Ok(combined_results {
                  results_video: a?,
                  results_image: b?,
                  hash: hash,
                  time: get_current_time(),
                  version: Version {
                      major: crate::version::major_version(),
                      minor: crate::version::minor_version(),
                      patch: crate::version::patch_version(),
                  },
                  key: key,
              });
          } else {
              return Err(anyhow::format_err!("Not implemented"));
          }
      }

      pub async fn video_file_2_infer(

      pub async fn video_file_2_infer_file(
          &self,
          mut inpath: std::path::PathBuf,
      ) -> anyhow::Result<()> {
          let buffer: Vec<u8> = {
              let res = self.video_file_2_infer(&inpath).await?.to_proto();
              let mut tmp = Vec::with_capacity(res.encoded_len());
              res.encode(&mut tmp)?;
              tmp
          };

          if inpath.add_extension("proto") {
              tokio::fs::write(inpath, buffer).await?;
              return Ok(());
          } else {
              return Err(anyhow::format_err!(
                  "Failed to create the proto file extension"
              ));
          }
      }

      pub async fn video_dir_2_infer_dir(
          &self,
          inpath: impl AsRef<std::path::Path>,
      ) -> anyhow::Result<()> {
          let walker: Vec<std::path::PathBuf> = jwalk::WalkDir::new(inpath)
              .into_iter()
              .flatten()
              .map(|i| -> Option<std::path::PathBuf> {
                  match i.path().extension() {
                      None => {
                          return None;
                      }
                      Some(e) => {
                          if e == "mp4" {
                              return Some(i.path().to_owned());
                          } else {
                              return None;
                          }
                      }
                  }
              })
              .flatten()
              .collect();

          let _ = futures::stream::iter(walker.into_iter().map(|i| self.video_file_2_infer_file(i)))
              .buffer_unordered(8)
              .collect::<Vec<_>>()
              .await
              .into_iter()
              .for_each(|e| match e {
                  Err(e) => {
                      tracing::error!("Failed inference for file {}", e)
                  }
                  Ok(_) => {}
              });

          Ok(())
      }

      pub async fn video_2_fft(
          &self,
          main_input: crate::serverinferencechannelvideodecoder::MainInput,
      ) -> anyhow::Result<Vec<crate::serverinferencechannelfft::MainOutput>> {
          let video_tensor = self.video_decoder.main_fn(main_input).await??;
          let video_view = crate::videoview::video_slicer_mapped::new(video_tensor.data)?;
          let windowed_slices = video_view.get_windowed_slices()?;
          let slave_fft = self.video_fft.clone();

          let res = futures::stream::iter(windowed_slices.into_iter().map(|i| slave_fft.main_fn(i)))
              .buffer_unordered(8)
              .collect::<Vec<_>>()
              .await
              .into_iter()
              .flatten()
              .collect::<Vec<_>>();

          return Ok(res);
      }

      pub async fn video_file_2_fft_file(
          &self,
          inpath: impl AsRef<std::path::Path>,
      ) -> anyhow::Result<()> {
          tracing::info!("Working on input video file {:?}", inpath.as_ref());

          let outdir = {
              let mut tmp = inpath.as_ref().to_owned();
              tmp.add_extension("dir");
              tmp
          };

          tokio::fs::create_dir_all(&outdir).await?;
          tracing::info!("Done creating directory");

          let outpathstart = {
              let mut tmp = outdir.clone();
              tmp.push("start");
              tmp
          };

          let outpathdone = {
              let mut tmp = outdir.clone();
              tmp.push("done");
              tmp
          };

          let inprogress: bool = {
              tracing::info!("Checking if the task is already taken");
              match tokio::fs::try_exists(&outpathstart).await {
                  Ok(o) => o,
                  Err(_) => false,
              }
          };

          let existance: bool = {
              tracing::info!("Checking if the task is already done");
              match tokio::fs::try_exists(&outpathdone).await {
                  Ok(o) => o,
                  Err(_) => false,
              }
          };

          if (existance || inprogress) {
              tracing::info!("The done or start signature already exists, not doing anything");
              return Ok(());
          } else {
              tokio::fs::write(&outpathstart, "").await?;

              let res = tokio::fs::read(&inpath).await?;

              tracing::info!("Done reading the video file");

              let outdir = std::sync::Arc::new(outdir);

              let frames = self.video_2_fft(res).await?;

              futures::stream::iter(frames.iter().flatten().enumerate())
                  .for_each_concurrent(Some(8), |(index, i)| {
                      let outdir = outdir.clone();
                      async move {
                          let fft_item = crate::videofftbf16::fft_video::from_32(i);

                          let mut outpath = (*outdir).clone();
                          outpath.push(index.to_string());
                          outpath.set_extension("bin16");

                          tracing::info!("writing file {:?}", outpath);
                          if let Err(e) = fft_item.save(&outpath).await {
                              tracing::error!("Failed to save file {:?}: {}", outpath, e);
                          }
                      }
                  })
                  .await;

              tokio::fs::write(outpathdone, "").await?;

              return Ok(());
          }
      }

      pub async fn video_dir_2_ffts(
          &self,
          path_dir_input: impl AsRef<std::path::Path>,
      ) -> anyhow::Result<()> {
          let walker = jwalk::WalkDir::new(path_dir_input)
              .into_iter()
              .filter_map(|e| e.ok())
              .map(|e| e.path())
              .filter_map(|e| {
                  let ext = e.extension();
                  match ext {
                      None => None,
                      Some(o) => {
                          if o == "mp4" {
                              Some(e)
                          } else {
                              None
                          }
                      }
                  }
              });

          tracing::info!("Obtained iterator of files from dir");

          let _ = futures::stream::iter(walker)
              .for_each_concurrent(Some(8), |i| async move {
                  let _ = self.video_file_2_fft_file(i).await;
              })
              .await;

          // println!("{:?}", walker);

          Ok(())
      }
  }

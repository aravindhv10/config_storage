* Base image

** Main Dockerfile
#+begin_src conf :mkdirp yes :tangle ./Dockerfile
  FROM docker.io/nixos/nix
  RUN nix-channel --update
#+end_src

** Build script
#+begin_src sh :shebang #!/bin/sh :results output :mkdirp yes :tangle ./build_container.sh
  cd "$('dirname' -- "${0}")"
  buildah build --tag nixbuilder .
#+end_src

** run script
#+begin_src sh :shebang #!/bin/sh :results output :mkdirp yes :tangle ./run_container.sh
  cd "$('dirname' -- "${0}")"
  podman run \
      '--rm' \
      '--interactive' \
      '--tty' \
      -v "$(realpath .):/data" \
      -v "$(realpath "${HOME}/GITHUB"):/root/GITHUB" \
      'nixbuilder' \
      'bash' ;
#+end_src

* Important configs
#+begin_src conf :mkdirp yes :tangle ./Dockerfile
  USER root
  ENV HOME=/root
  WORKDIR /root
#+end_src

* The nix shell

** Copy the code to docker and run
#+begin_src conf :mkdirp yes :tangle ./Dockerfile
  COPY shell.nix /root/shell.nix
#+end_src

** Actual code
#+begin_src conf :mkdirp yes :tangle ./shell.nix
  {pkgs ? import <nixpkgs> {}}: let
    mylist = with pkgs; [
      cargo
      cargo-zigbuild
      git
      rustc
      rustup
      zig
    ];
  in (pkgs.mkShell {
    name = "good_rust_zig_env";
    packages = mylist;
    runScript = "ls";
  })
#+end_src

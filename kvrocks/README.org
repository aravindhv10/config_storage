* Docker scripts for postgres

** Configuration for the server
#+begin_src sh :shebang #!/bin/sh :results output :mkdirp yes :tangle ./env.sh
  mkdir -p -- './volume'
  NAME_IMAGE='docker.io/apache/kvrocks'
  NAME_GUEST='container_kvrocks'
  PORT_GUEST='6666'
  PORT_HOST='8102'
  VOLUME_GUEST='/var/lib/kvrocks'
  VOLUME_HOST="$('realpath' './volume')"
#+end_src

** Shell script to start the server container
#+begin_src sh :shebang #!/bin/sh :results output :mkdirp yes :tangle ./run.sh
  cd "$(dirname -- "${0}")"

  . './env.sh'

  'podman' 'run' \
      '-it' '--rm' \
      '--name' "${NAME_GUEST}" \
      '--network' 'app_net' \
      '-p' "${PORT_HOST}:${PORT_GUEST}" \
      '-v' "${VOLUME_HOST}:${VOLUME_GUEST}" \
      "${NAME_IMAGE}" \
          '--bind' '0.0.0.0' \
          '--dir' '/var/lib/kvrocks'

  exit '0'
#+end_src
